    if(id==='fmt ')fmt={format:dv.getUint16(b,true),channels:dv.getUint16(b+2,true),sampleRate:dv.getUint32(b+4,true),bits:dv.getUint16(b+14,true)};
    else if(id==='data'){dOff=b;dLen=sz;}off=b+sz+(sz&1);}
  if(!fmt||!dOff)throw new Error('WAV: fmt/data ausentes');
  const ch=fmt.channels,by=fmt.bits/8,fr=Math.floor(dLen/(by*ch)),out=new Float32Array(fr);
  for(let i=0;i<fr;i++){const p=dOff+i*by*ch;
    if(fmt.format===3&&fmt.bits===32)out[i]=dv.getFloat32(p,true);
    else if(fmt.format===1&&fmt.bits===16)out[i]=dv.getInt16(p,true)/32768;
    else if(fmt.format===1&&fmt.bits===32)out[i]=dv.getInt32(p,true)/2147483648;
    else if(fmt.format===1&&fmt.bits===24){const a=dv.getUint8(p),x=dv.getUint8(p+1),y=dv.getUint8(p+2);let v=(y<<16)|(x<<8)|a;if(v&0x800000)v-=0x1000000;out[i]=v/8388608;}}
  return {sampleRate:fmt.sampleRate,channels:ch,bits:fmt.bits,format:fmt.format,frames:fr,durationSec:fr/fmt.sampleRate,samples:out};
}
function captureIntegrity(s){let peak=0,sq=0,sum=0,clip=0,sil=0;const n=s.length;
  for(let i=0;i<n;i++){const a=Math.abs(s[i]);if(a>peak)peak=a;sq+=s[i]*s[i];sum+=s[i];if(a>=0.999)clip++;if(a<1e-4)sil++;}
  const rms=Math.sqrt(sq/Math.max(1,n)),db=v=>v>0?20*Math.log10(v):-Infinity;
  return {peak,peakDb:db(peak),rms,rmsDb:db(rms),crest:db(peak)-db(rms),dc:sum/Math.max(1,n),clip,silPct:100*sil/Math.max(1,n)};}

async function fromAamp(file){

  const an=man.analysis&&f[man.analysis]?JSON.parse(asText(f[man.analysis])):null;
  const caps=(man.capture||[]).map(c=>({stimulus:c.stimulus,file:c.file,wav:f[c.file]?parseWAV(asAB(f[c.file])):null})).filter(c=>c.wav);
  let pr=caps.findIndex(c=>c.file===man.primaryCapture);if(pr<0)pr=0;
  const meta=(an&&an.meta)||{};

    meta:{device:man.device||meta.device,sampleRate:man.sampleRate||meta.sampleRate,mode:man.mode||meta.mode,timestamp:man.timestamp||meta.timestamp},
    captures:caps,primary:pr,metrics:normalizeMetrics(an),hasAnalysis:!!an,acquisition:man.acquisition||null};}

/* Suavização fracional de oitava LOCAL, usada EXCLUSIVAMENTE por buildFrequencyEnv.
   Bit-idêntica a smoothFracOct (mesma janela, mesma ordem de soma crescente, mesmo filtro isFinite),
   porém O(n·W) por janela deslizante de dois ponteiros monotônicos — o eixo de frequência x é
   ascendente, então a janela em log2 [lx[i]-half, lx[i]+half] avança monotonicamente. Se x não for
   ascendente, cai no algoritmo O(n²) idêntico ao compartilhado. NÃO altera o helper compartilhado. */
function smoothFracOctFq(x,y,frac){if(!frac)return y.slice();
  const N=y.length,half=1/(2*frac),out=new Array(N),lx=new Array(N);
  let asc=true;for(let j=0;j<N;j++){lx[j]=Math.log2(x[j]);if(j&&lx[j]<lx[j-1])asc=false;}
  if(!asc){for(let i=0;i<N;i++){const lo=lx[i]-half,hi=lx[i]+half;let s=0,n=0;
      for(let j=0;j<N;j++){if(lx[j]>=lo&&lx[j]<=hi&&isFinite(y[j])){s+=y[j];n++;}}out[i]=n?s/n:y[i];}return out;}
  let jlo=0,jhi=-1;
  for(let i=0;i<N;i++){const lo=lx[i]-half,hi=lx[i]+half;
    while(jlo<N&&lx[jlo]<lo)jlo++;                       // 1ª posição com lx>=lo (monotônico)
    if(jhi<jlo-1)jhi=jlo-1;                              // jhi = última posição com lx<=hi
    while(jhi+1<N&&lx[jhi+1]<=hi)jhi++;
    let s=0,n=0;for(let j=jlo;j<=jhi;j++){if(isFinite(y[j])){s+=y[j];n++;}}  // soma fresca, ordem crescente
    out[i]=n?s/n:y[i];}
  return out;}

function detectPeaks(x,y,minProm,sign){const yy=sign>0?y:y.map(v=>-v),n=yy.length,out=[];
  for(let i=1;i<n-1;i++){if(!(yy[i]>yy[i-1]&&yy[i]>=yy[i+1]))continue;
    let l=i;while(l>0&&yy[l-1]<=yy[i])l--;let lmin=yy[i];for(let k=i;k>=l;k--)lmin=Math.min(lmin,yy[k]);
    let r=i;while(r<n-1&&yy[r+1]<=yy[i])r++;let rmin=yy[i];for(let k=i;k<=r;k++)rmin=Math.min(rmin,yy[k]);
    const prom=yy[i]-Math.max(lmin,rmin);if(prom<minProm)continue;
    const L=y[i],tgt=sign>0?L-3:L+3,below=sign>0;
    const fL=crossOut(x,y,i,tgt,-1,below),fR=crossOut(x,y,i,tgt,1,below);
    const bw=(fL&&fR)?fR-fL:null,Q=bw?x[i]/bw:null;
    out.push({f:x[i],level:y[i],prom,fL,fR,bw,Q,a:(fL||x[l]),b:(fR||x[r])});}
  out.sort((p,q)=>q.prom-p.prom);return out;}
function steepest(x,y){let bi=1,bs=0;for(let i=1;i<x.length-1;i++){const s=(y[i+1]-y[i-1])/(Math.log2(x[i+1])-Math.log2(x[i-1]));if(Math.abs(s)>Math.abs(bs)){bs=s;bi=i;}}
  return {f:x[bi],slope:bs,a:x[Math.max(0,bi-6)],b:x[Math.min(x.length-1,bi+6)]};}
function rangeStats(curve,a,b){const x=curve.x,y=curve.y;let mn=Infinity,mx=-Infinity,sum=0,n=0,sx=0,sy=0,sxx=0,sxy=0,pkI=-1,ntI=-1;
  for(let i=0;i<x.length;i++){if(x[i]<a||x[i]>b||!isFinite(y[i]))continue;if(y[i]<mn){mn=y[i];ntI=i;}if(y[i]>mx){mx=y[i];pkI=i;}sum+=y[i];const lx=Math.log2(x[i]);n++;sx+=lx;sy+=y[i];sxx+=lx*lx;sxy+=lx*y[i];}
  if(!n)return null;const slope=n>1?(n*sxy-sx*sy)/(n*sxx-sx*sx):null;
  return {n,min:mn,max:mx,mean:sum/n,ripple:mx-mn,slope,pkF:pkI>=0?x[pkI]:null,ntF:ntI>=0?x[ntI]:null};}

function etcToDb(y){let mx=0;for(const v of y){const a=Math.abs(v);if(a>mx)mx=a;}mx=mx||1;
  return y.map(v=>{const d=20*Math.log10(Math.abs(v)/mx);return isFinite(d)?Math.max(d,-120):-120;});}
function schroeder(x,y){const idx=[];for(let i=0;i<x.length;i++)if(x[i]>=0)idx.push(i);
  const N=idx.length,e=new Array(N);let cum=0;for(let k=N-1;k>=0;k--){const v=y[idx[k]];cum+=v*v;e[k]=cum;}
  const e0=e[0]||1,xs=idx.map(i=>x[i]),ys=e.map(v=>{const d=10*Math.log10(v/e0);return isFinite(d)?Math.max(d,-120):-120;});return {x:xs,y:ys};}
function crossTime(xs,ys,tgt){for(let i=1;i<xs.length;i++){if((ys[i-1]-tgt)*(ys[i]-tgt)<=0&&ys[i-1]!==ys[i]){const t=(tgt-ys[i-1])/(ys[i]-ys[i-1]);return xs[i-1]+t*(xs[i]-xs[i-1]);}}return null;}
function linFitRange(xs,ys,yHi,yLo){let n=0,sx=0,sy=0,sxx=0,sxy=0;for(let i=0;i<xs.length;i++){const v=ys[i];if(v<=yHi&&v>=yLo){n++;sx+=xs[i];sy+=v;sxx+=xs[i]*xs[i];sxy+=xs[i]*v;}}return n>1?(n*sxy-sx*sy)/(n*sxx-sx*sx):null;}
function timeStats(x,y,a,b){let pk=0,sq=0,n=0,pi=-1;for(let i=0;i<x.length;i++){if(x[i]<a||x[i]>b)continue;const av=Math.abs(y[i]);if(av>pk){pk=av;pi=i;}sq+=y[i]*y[i];n++;}return n?{peak:pk,peakT:x[pi],rms:Math.sqrt(sq/n),energy:sq,n}:null;}

/* MS-04: o engine emite APENAS a curva de IR ("Impulse Response"); a curva de envelope pode não
   vir no payload. Este helper devolve {ir, env}: usa o envelope explícito se existir (ex.: fixtures
   .aamp), senão DERIVA um envelope de exibição a partir da própria IR — mesma prática já usada no
   Laboratório para Schroeder/ETC (curvas derivadas da IR para representação). NÃO altera métrica,
   DSP ou payload; é apenas a representação da bancada temporal a partir do dado disponível.
   Regex ampliadas para casar nomenclatura EN ("Impulse Response") e PT ("impulso"/"envoltória"). */
function ms04Curves(m04){
  if(!m04||!m04.curves)return {ir:null,env:null};
  const ir=m04.curves.find(c=>/impuls/i.test(c.name))||null;
  let env=m04.curves.find(c=>/envelope|envolt/i.test(c.name))||null;
  if(!env&&ir&&ir.x&&ir.x.length){
    const y=ir.y,n=y.length,mag=new Array(n);
    for(let i=0;i<n;i++)mag[i]=Math.abs(y[i]);
    const w=Math.max(1,Math.round(n/600)),sm=new Array(n);   // envelope: |IR| com suavização curta
    let acc=0;for(let i=0;i<Math.min(n,w+1);i++)acc+=mag[i];  // média móvel O(n)
    for(let i=0;i<n;i++){const lo=Math.max(0,i-w),hi=Math.min(n-1,i+w);
      if(i>0){const add=i+w;if(add<n)acc+=mag[add];const rem=i-w-1;if(rem>=0)acc-=mag[rem];}
      sm[i]=acc/(hi-lo+1);}
    env={name:'Envelope (derivado da IR)',xUnit:ir.xUnit||'ms',yUnit:'',x:ir.x.slice(),y:sm,derived:true};
  }
  return {ir,env};
}

  comb:{t:'Pente-filtro',d:'Uma reflexão atrasada Δt soma-se ao direto e cria notches em (2k−1)/(2Δt). Origem temporal — EQ não resolve; trate a reflexão.'},

  itdg:{t:'ITDG',d:'Initial Time Delay Gap — intervalo entre o som direto e a 1ª reflexão; define a sensação de proximidade e o início do pente-filtro.'},
  thd:{t:'THD / harmônicos',d:'Energia em múltiplos da fundamental. Acima de ~1% indica não-linearidade audível; a assinatura par/ímpar caracteriza sua natureza.'},

/* Correlation Engine — dado um fenômeno num domínio, encontra seus correlatos nos demais */
function correlate(ph){const out=[],phc=M.metrics.curves['MS-02'],gd=M.metrics.curves['MS-03'],m04=M.metrics.byId&&M.metrics.byId['MS-04'];
  if(ph.dom==='frequency'&&ph.f){const f=ph.f;
    if(phc){const rs=rangeStats(phc,f*0.8,f*1.25);if(rs&&rs.ripple>20)out.push({dom:'phase',label:'Coincide em fase',detail:fmt(rs.ripple,0)+'° em torno de '+fHz(f)+' — fase mínima'});}
    if(gd){const s=gd.y.filter(isFinite).slice().sort((a,b)=>a-b),med=s[s.length>>1],ex=gd.y[nearestIdx(gd.x,f)]-med;if(Math.abs(ex)>0.5)out.push({dom:'groupdelay',label:'Pico de atraso de grupo',detail:fmt(ex,2)+' ms de excesso em '+fHz(f)});}
    if(ph.type==='notch'&&m04){const env=ms04Curves(m04).env;if(env){const etc=etcToDb(env.y),refl=detectPeaks(env.x,etc,6,1).filter(p=>p.f>1.5&&p.f<50),dt=500/f,near=refl.sort((a,b)=>Math.abs(a.f-dt)-Math.abs(b.f-dt))[0];if(near&&Math.abs(near.f-dt)<dt*0.6)out.push({dom:'impulse',label:'Compatível com reflexão',detail:'reflexão em '+fmt(near.f,2)+' ms prevê este notch'});}}}
  if(ph.dom==='impulse'&&ph.t)out.push({dom:'frequency',label:'Imprime pente-filtro',detail:'notches a partir de '+fHz(500/ph.t)});
  return out;}

/* Recommendation Engine — opções e interpretação (evidência, não veredicto) */
function recommend(a){const l=a.label||'';
  if(a.dom==='frequency'){if(/Ressonância/.test(l)){const f=a.focus&&a.focus.f;return {opt:'Opção de correção',text:'Coincidindo em fase, é de fase mínima: um bell invertido em '+(f?fHz(f):'na frequência')+' (mesma Q) achata o pico — confirme antes que não mascara uma reflexão.'};}
    if(/Notch/.test(l))return {opt:'Interpretação',text:'Notch estreito sugere cancelamento por reflexão; EQ aditivo não resolve — investigue a origem temporal na Impulsiva.'};
    if(/atraso de grupo/i.test(l))return {opt:'Interpretação',text:'Variação de GD acima de ~1 ms é audível como smearing; comum perto de crossovers/allpass — verifique alinhamento.'};
    if(/Roll-?off/i.test(l))return {opt:'Interpretação',text:'Banda útil termina cedo — limite do sistema ou do estímulo; confirme a faixa esperada.'};
    if(/Inclinação/.test(l))return {opt:'Interpretação',text:'Inclinação espectral sugere balanço tonal geral; um shelf amplo equilibra sem tocar fenômenos locais.'};}
  if(a.dom==='impulse'&&/Reflexão/.test(l)){const t=a.focus&&a.focus.t;return {opt:'Opção de tratamento',text:'Reflexão'+(t?' a '+fmt(t,2)+' ms':'')+' gera pente-filtro em '+(t?fHz(500/t):'')+' e múltiplos — absorção/reposicionamento reduz na origem; EQ não.'};}

  if(a.dom==='imd')return {opt:'Interpretação',text:'Distorção elevada indica não-linearidade; reduza o nível de excitação e reavalie para separar comportamento de sobrecarga.'};
  if(a.dom==='noise')return {opt:'Interpretação',text:'Piso/SNR limitam a menor variação confiável; blindagem e ganho de entrada ampliam a janela útil.'};
  return null;}

function fftRadix2(re,im){const n=re.length;for(let i=1,j=0;i<n;i++){let bit=n>>1;for(;j&bit;bit>>=1)j^=bit;j^=bit;if(i<j){const tr=re[i];re[i]=re[j];re[j]=tr;const ti=im[i];im[i]=im[j];im[j]=ti;}}
  for(let len=2;len<=n;len<<=1){const ang=-2*Math.PI/len,wr=Math.cos(ang),wi=Math.sin(ang);
    for(let i=0;i<n;i+=len){let cr=1,ci=0;for(let k=0;k<len/2;k++){const a=i+k,b=i+k+len/2,vr=re[b]*cr-im[b]*ci,vi=re[b]*ci+im[b]*cr;
      re[b]=re[a]-vr;im[b]=im[a]-vi;re[a]+=vr;im[a]+=vi;const ncr=cr*wr-ci*wi;ci=cr*wi+ci*wr;cr=ncr;}}}}

  for(let i=0;i<L;i++){const w=0.5-0.5*Math.cos(2*Math.PI*i/(L-1));re[i]=frame[i]*w;}fftRadix2(re,im);
  const half=N>>1,f=new Array(half),y=new Array(half);for(let k=0;k<half;k++){f[k]=k*fs/N;y[k]=20*Math.log10(Math.hypot(re[k],im[k])/N*2+1e-12);}return {f,y};}
function stft(samples,fs,frameSize,hop){const N=frameSize,bins=N>>1,freqs=new Array(bins),frames=[];
  for(let k=0;k<bins;k++)freqs[k]=k*fs/N;const win=new Array(N);for(let i=0;i<N;i++)win[i]=0.5-0.5*Math.cos(2*Math.PI*i/(N-1));
  for(let s=0;s+N<=samples.length;s+=hop){const re=new Array(N),im=new Array(N).fill(0);for(let i=0;i<N;i++)re[i]=samples[s+i]*win[i];fftRadix2(re,im);
    const col=new Float32Array(bins);for(let k=0;k<bins;k++)col[k]=20*Math.log10(Math.hypot(re[k],im[k])/N*2+1e-12);frames.push({t:(s+N/2)/fs*1000,col});}

/* =========================================================================
   AAMP — um instrumento, quatro estados: Preparação → Medição → Processamento
   → Laboratório (Investigação, = #view, reuso integral). O frontend só conduz
   o fluxo e representa estados; o Measurement Engine (C++) executa a lógica.
   A ponte é in-process (WebView); abaixo há um transporte SIMULADO que replica
   a sequência do MeasurementOrchestrator, trocável pelo binding nativo sem
   alterar a casca. Filosofia: uma única Session viva; sem histórico/persistência.
   ========================================================================= */
const AAMP=(function(){
  const el=id=>document.getElementById(id);
  const listeners={};
  function on(evt,cb){(listeners[evt]||(listeners[evt]=[])).push(cb);}
  function emit(evt,data){(listeners[evt]||[]).forEach(cb=>{try{cb(data);}catch(e){console.error(e);}});}
  let transport=null;
  function send(cmd,args){if(transport)transport.command(cmd,args||{});}

  /* ---- Transporte nativo (WebView) se o host o expuser; senão, mock ---- */
  function nativeTransport(){
    // JUCE 8 WebBrowserComponent — integração nativa (withNativeIntegrationEnabled)
    if(window.__JUCE__&&window.__JUCE__.backend){
      try{
        window.__JUCE__.backend.addEventListener('aampEvent',pl=>{const d=(typeof pl==='string')?JSON.parse(pl):pl;emit(d.evt,d.data);});
        // Comando JS->C++ pela API documentada do JUCE 8: backend.emitEvent(eventId, payload).
        // (getNativeFunction so existe se o modulo JS index.js do JUCE for carregado; o shim nativo
        //  sozinho nao o expoe — por isso usamos o canal backend, ja disponivel e usado acima.)
        return {name:'native-juce',command:(cmd,args)=>window.__JUCE__.backend.emitEvent('aampCommand',JSON.stringify({cmd,args:args||{}}))};
      }catch(e){}
    }
    // Ponte genérica: o host injeta window.aampNative.postMessage e chama window.aampEvent(evt,data)
    if(window.aampNative&&window.aampNative.postMessage){
      window.aampEvent=(evt,data)=>emit(evt,(typeof data==='string')?JSON.parse(data):data);
      return {name:'native',command:(cmd,args)=>window.aampNative.postMessage(JSON.stringify({cmd,args:args||{}}))};
    }
    return null;
  }

  /* ---- MockEngine: replica a sequência do MeasurementOrchestrator ---- */
  function synthPayload(){
    const N=256,x=[],y=[];for(let i=0;i<N;i++){const f=20*Math.pow(1000,i/(N-1));x.push(f);
      const g=9.5*Math.exp(-Math.pow(Math.log(f/2840)/0.28,2));y.push(g-(f>5000?3.2*Math.log(f/5000):0));}
    const cap=[];for(let i=0;i<2048;i++)cap.push((i===40?1:0)+(Math.random()*2-1)*0.02*Math.exp(-i/300));
    return {source:'Measurement Engine (mock)',
      meta:{device:'Loopback Virtual',sampleRate:48000,mode:'Mono',timestamp:new Date().toISOString()},
      metrics:[{id:'MS-01',name:'Resposta em frequência',quality:'Normal',
        scalars:[{key:'passband_lo',value:1020,unit:'Hz',quality:'Normal',valuePresent:true},
                 {key:'passband_hi',value:5370,unit:'Hz',quality:'Normal',valuePresent:true}],
        curves:[{name:'Magnitude',xUnit:'Hz',yUnit:'dB',x,y}]}],
      captures:[{stimulus:'sweep',samples:cap,sampleRate:48000,channels:1}],primary:0};
  }
  const MockEngine=(function(){
    let cancelled=false,timers=[],payload=synthPayload();
    const T=(fn,ms)=>{timers.push(setTimeout(fn,ms));};
    const clear=()=>{timers.forEach(clearTimeout);timers=[];};
    function command(cmd,args){
      if(cmd==='enumerate')T(()=>emit('devices',{devices:[{id:'loopback',name:'Loopback Virtual',backend:'Mock',in:2,out:2}]}),100);
      else if(cmd==='prepare'){cancelled=false;T(()=>emit('prepared',{state:'Ready',
        caps:{backend:'Mock',inputChannels:2,outputChannels:2,sampleRates:[44100,48000,96000],bufferSizes:[128,256,512],bitDepths:[24],manufacturer:''},
        operating:{sampleRate:48000},
        validation:{ok:true,usableChannels:true,rateSupported:true,latencyMs:10.7,latencyPartial:true}}),160);}
      else if(cmd==='start')run();
      else if(cmd==='cancel'){cancelled=true;clear();T(()=>{emit('state',{state:'Idle'});emit('runFinished',{outcome:'Cancelled'});},120);}
      else if(cmd==='export')T(()=>emit('exported',{ok:true,format:(args&&args.format)||'pdf'}),300);
    }
    function run(){
      cancelled=false;const N=8,step=160;let t=0;
      emit('state',{state:'Measuring'});
      for(let i=0;i<N;i++){T(()=>{if(cancelled)return;emit('progress',{label:'Gerando e capturando estímulos',percent:Math.round((i+0.5)/N*70),completed:i,total:N});emit('stimulusIntegrity',{index:i,ok:true});},t+=step);}
      T(()=>{if(cancelled)return;emit('state',{state:'Processing'});emit('progress',{label:'Calculando métricas oficiais',percent:76});},t+=step);
      const mm=[['MS-01','Resposta em frequência'],['MS-04','Resposta ao impulso'],['MS-07','Distorção harmônica'],['MS-11','Ruído & piso']];
      mm.forEach((m,k)=>T(()=>{if(cancelled)return;emit('metricReady',{id:m[0],name:m[1]});emit('progress',{label:'Calculando métricas oficiais',percent:76+Math.round((k+1)/mm.length*22)});},t+=step));
      T(()=>{if(cancelled)return;emit('progress',{label:'Caracterização concluída',percent:100});emit('runFinished',{outcome:'Completed',payload});},t+=step);
    }
    return {name:'mock',command,setPayload:p=>{payload=p;}};
  })();

  transport=nativeTransport()||MockEngine;

  /* ---- Estado do produto: um ambiente, dois estados (sem/com Session) ---- */
